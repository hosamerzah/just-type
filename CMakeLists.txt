cmake_minimum_required(VERSION 3.24)
project(AutoKeyboard VERSION 0.2.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB_RECURSE AK_ID_INPUTS CONFIGURE_DEPENDS src/* scripts/* tests/* assets/* packs/*)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${AK_ID_INPUTS})
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/build_identity.py"
  --header "${CMAKE_BINARY_DIR}/generated/build_identity.hpp" --version "${PROJECT_VERSION}"
  COMMAND_ERROR_IS_FATAL ANY)
file(STRINGS "${CMAKE_BINARY_DIR}/generated/build_identity.hpp" AK_ID_LINE REGEX "AK_SOURCE_ID")
string(REGEX REPLACE ".*\"([a-f0-9]+)\".*" "\\1" AK_SOURCE_ID "${AK_ID_LINE}")
include_directories("${CMAKE_BINARY_DIR}/generated")
if(NOT WIN32)
  message(FATAL_ERROR "AutoKeyboard requires Windows.")
endif()
set(DEPS "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "Bootstrapped dependencies")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(AK_ARCH x64)
else()
  set(AK_ARCH x86)
endif()
set(PDFIUM "${DEPS}/pdfium-${AK_ARCH}")
if(NOT EXISTS "${DEPS}/hunspell/src/hunspell/hunspell.hxx")
  message(FATAL_ERROR "Run python scripts/bootstrap.py first.")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/utf-8 /W4 /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wno-unknown-pragmas)
  add_link_options(-static)
endif()
file(GLOB HUNSPELL_SOURCES "${DEPS}/hunspell/src/hunspell/*.cxx")
add_library(hunspell STATIC ${HUNSPELL_SOURCES})
target_include_directories(hunspell PUBLIC "${DEPS}/hunspell/src/hunspell")
target_compile_definitions(hunspell PUBLIC HUNSPELL_STATIC PRIVATE BUILDING_LIBHUNSPELL)
add_library(sqlite STATIC "${DEPS}/sqlite/sqlite3.c")
target_include_directories(sqlite PUBLIC "${DEPS}/sqlite")
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
add_library(keyboard_core STATIC src/text.cpp src/engine.cpp src/store.cpp src/runtime.cpp src/global_hook.cpp)
target_include_directories(keyboard_core PUBLIC src)
target_link_libraries(keyboard_core PUBLIC hunspell sqlite normaliz shlwapi shell32 ole32 advapi32 user32)
target_include_directories(keyboard_core PUBLIC assets)
add_library(pdf_import STATIC src/pdf_import.cpp)
add_library(pdf_worker_client STATIC src/pdf_worker_client.cpp)
target_link_libraries(pdf_worker_client PUBLIC keyboard_core)
add_executable(AutoKeyboardPdfWorker src/pdf_worker.cpp assets/app.rc)
target_link_libraries(AutoKeyboardPdfWorker PRIVATE pdf_import)
add_custom_command(TARGET AutoKeyboardPdfWorker POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/packs" "$<TARGET_FILE_DIR:AutoKeyboardPdfWorker>/packs")
if(MINGW)
  target_link_options(AutoKeyboardPdfWorker PRIVATE -municode)
endif()
add_library(keyboard_profile STATIC src/profile.cpp)
target_link_libraries(keyboard_profile PUBLIC keyboard_core uuid)
add_executable(AutoKeyboardControl src/control.cpp assets/app.rc)
target_link_libraries(AutoKeyboardControl PRIVATE keyboard_profile)
add_executable(AutoKeyboardNotepadDiagnostic src/control.cpp assets/app.rc)
add_executable(AutoKeyboardTypingTest WIN32 src/diagnostic_host.cpp assets/app.rc)
target_link_libraries(AutoKeyboardTypingTest PRIVATE keyboard_profile user32 uuid)
if(MINGW)
  target_link_options(AutoKeyboardTypingTest PRIVATE -municode)
endif()
target_compile_definitions(AutoKeyboardNotepadDiagnostic PRIVATE AK_TRACE_NOTEPAD_DEFAULT)
target_link_libraries(AutoKeyboardNotepadDiagnostic PRIVATE keyboard_profile user32)
if(MINGW)
  target_link_options(AutoKeyboardControl PRIVATE -municode)
  target_link_options(AutoKeyboardNotepadDiagnostic PRIVATE -municode)
endif()
target_include_directories(pdf_import PRIVATE "${PDFIUM}/include")
target_link_libraries(pdf_import PUBLIC keyboard_core "${PDFIUM}/lib/pdfium.dll.lib")
add_executable(AutoKeyboard WIN32 src/app.cpp assets/app.rc)
if(MINGW)
  target_link_options(AutoKeyboard PRIVATE -municode)
endif()
target_link_libraries(AutoKeyboard PRIVATE pdf_worker_client keyboard_profile comctl32 comdlg32 user32 gdi32)
add_dependencies(AutoKeyboard AutoKeyboardPdfWorker)
add_library(AutoKeyboardIME SHARED src/tsf.cpp src/registration.cpp src/ime.def assets/app.rc)
target_compile_definitions(AutoKeyboardIME PRIVATE AK_IME_BINARY)
set_target_properties(AutoKeyboardIME PROPERTIES PREFIX "")
target_link_libraries(AutoKeyboardIME PRIVATE keyboard_core uuid advapi32 user32 gdi32)
if(MSVC)
  target_link_libraries(AutoKeyboardIME PRIVATE msctf)
endif()
add_executable(keyboard_tests tests/tests.cpp)
target_link_libraries(keyboard_tests PRIVATE pdf_import pdf_worker_client)
add_dependencies(keyboard_tests AutoKeyboardPdfWorker)
add_executable(keyboard_tsf_tests tests/tsf_tests.cpp)
target_link_libraries(keyboard_tsf_tests PRIVATE keyboard_core keyboard_profile uuid user32)
add_dependencies(keyboard_tsf_tests AutoKeyboardIME)
add_custom_command(TARGET keyboard_tsf_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/packs" "$<TARGET_FILE_DIR:keyboard_tsf_tests>/packs")
enable_testing()
add_executable(keyboard_setup_tests tests/setup_tests.cpp assets/app.rc)
target_include_directories(keyboard_setup_tests PRIVATE src assets)
add_test(NAME installer_state_checks COMMAND keyboard_setup_tests)
add_test(NAME keyboard_behavior COMMAND keyboard_tests)
add_test(NAME windows_tsf_integration COMMAND keyboard_tsf_tests)
add_executable(keyboard_ui_tests tests/ui_tests.cpp)
target_link_libraries(keyboard_ui_tests PRIVATE keyboard_core user32)
add_dependencies(keyboard_ui_tests AutoKeyboard)
add_test(NAME settings_ui_integration COMMAND keyboard_ui_tests)
set_tests_properties(windows_tsf_integration settings_ui_integration PROPERTIES RUN_SERIAL TRUE TIMEOUT 300 RESOURCE_LOCK desktop)
add_test(NAME build_tool_checks COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_build_tools.py")
set_tests_properties(installer_state_checks keyboard_behavior build_tool_checks PROPERTIES TIMEOUT 300)
set_tests_properties(keyboard_behavior PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
foreach(target AutoKeyboard AutoKeyboardPdfWorker keyboard_tests)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${PDFIUM}/bin/pdfium.dll" "$<TARGET_FILE_DIR:${target}>")
endforeach()
install(TARGETS AutoKeyboard AutoKeyboardIME AutoKeyboardControl AutoKeyboardPdfWorker AutoKeyboardNotepadDiagnostic AutoKeyboardTypingTest RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${PDFIUM}/bin/pdfium.dll" DESTINATION .)
install(DIRECTORY packs DESTINATION .)
install(DIRECTORY "${DEPS}/licenses/" DESTINATION licenses)
install(FILES README.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES tests/browser-fields.html DESTINATION diagnostics)
install(CODE "execute_process(COMMAND \"${Python3_EXECUTABLE}\" \"${CMAKE_SOURCE_DIR}/scripts/build_identity.py\" --stage \"\${CMAKE_INSTALL_PREFIX}\" --arch ${AK_ARCH} --version ${PROJECT_VERSION} --expected-source ${AK_SOURCE_ID} COMMAND_ERROR_IS_FATAL ANY)")
set(AK_SETUP_RC "" CACHE FILEPATH "Generated installer payload resource")
if(AK_SETUP_RC)
  add_executable(AutoKeyboardSetup WIN32 src/setup.cpp src/text.cpp "${AK_SETUP_RC}")
  target_include_directories(AutoKeyboardSetup PRIVATE src)
  target_link_libraries(AutoKeyboardSetup PRIVATE keyboard_profile shell32 ole32 uuid advapi32 bcrypt normaliz)
  if(MINGW)
    target_link_options(AutoKeyboardSetup PRIVATE -municode)
  endif()
endif()
